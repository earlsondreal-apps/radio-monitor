import sys import requests import os from PyQt5.QtWidgets import QApplication, QMainWindow from PyQt5.QtWebEngineWidgets import QWebEngineView from PyQt5.QtCore import QTimer # THE UI (HTML/CSS) - Removed localStorage logic HTML_CODE = """
Disconnected
Radio Monitor Pro
--.-
Waiting for Data
Enter IP:Port/file.json below
Hardware Address
""" class RadioApp(QMainWindow): def __init__(self): super().__init__() self.setWindowTitle("Radio Monitor") self.setFixedSize(400, 480) # Setup Browser self.browser = QWebEngineView() self.browser.setHtml(HTML_CODE) self.setCentralWidget(self.browser) # Config file path self.config_path = os.path.join(os.path.dirname(__file__), "config.txt") # Load saved address and put it in the input field once the page loads self.browser.loadFinished.connect(self.load_saved_address) # Timer to fetch data every 3 seconds self.timer = QTimer() self.timer.timeout.connect(self.run_cycle) self.timer.start(3000) def load_saved_address(self): if os.path.exists(self.config_path): with open(self.config_path, "r") as f: addr = f.read().strip() # Inject the saved address into the HTML input box self.browser.page().runJavaScript(f"document.getElementById('addr').value = '{addr}';") def run_cycle(self): # 1. Get whatever is currently typed in the input box self.browser.page().runJavaScript("document.getElementById('addr').value", self.fetch_data) def fetch_data(self, addr): if not addr: return # 2. Save the address to config.txt so it's remembered next time with open(self.config_path, "w") as f: f.write(addr) # 3. Fetch from hardware try: r = requests.get(f"http://{addr}", timeout=2) if r.status_code == 200: # Send data to HTML self.browser.page().runJavaScript(f"updateUI({r.text})") else: self.browser.page().runJavaScript("updateUI({'error': true})") except: self.browser.page().runJavaScript("updateUI({'error': true})") if __name__ == '__main__': app = QApplication(sys.argv) window = RadioApp() window.show() sys.exit(app.exec_())
